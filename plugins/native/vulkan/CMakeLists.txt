cmake_minimum_required(VERSION 3.24)
project(OpticaVulkan LANGUAGES CXX)
find_package(Vulkan REQUIRED COMPONENTS glslc)
add_library(optica_vk SHARED engine.cpp)
target_compile_features(optica_vk PRIVATE cxx_std_17)
set_property(TARGET optica_vk PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(optica_vk PRIVATE Vulkan::Vulkan)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/lens.spv COMMAND Vulkan::glslc ${CMAKE_CURRENT_SOURCE_DIR}/lens.comp -O --target-env=vulkan1.1 -o ${CMAKE_CURRENT_BINARY_DIR}/lens.spv DEPENDS lens.comp)
add_custom_target(lens_shader ALL DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/lens.spv)
add_dependencies(optica_vk lens_shader)
